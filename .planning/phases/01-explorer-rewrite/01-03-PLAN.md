### 1. calculateClientMatchScore function

Since `matching.ts` expects Prisma types (not available client-side from raw API data), write an inline simplified scoring function that computes a match score from the raw GitHub issue data:

```typescript
function calculateClientMatchScore(
  issue: GitHubIssue,
  preferredLanguages: string[],
  preferredTopics: string[],
  experienceLevel: string
): number {
  let score = 0;
  const repoName = extractRepoFromUrl(issue.repository_url);
  const issueLabels = issue.labels.map(l => l.name);
  const titleAndBody = issue.title.toLowerCase();

  // 1. Language match (max 25 pts)
  // Check if any preferred language likely matches based on labels/repo context
  // Since the raw API doesn't give us repo language directly in search results,
  // we check if label names or repo name hints contain language keywords
  const langMatch = preferredLanguages.some(lang =>
    repoName.toLowerCase().includes(lang.toLowerCase()) ||
    issueLabels.some(l => l.toLowerCase().includes(lang.toLowerCase()))
  );
  if (langMatch) score += 25;

  // 2. Interest/Topic match (max 25 pts)
  const interestMatch = preferredTopics.some(topic => {
    const t = topic.toLowerCase();
    return issueLabels.some(l => l.toLowerCase().includes(t)) ||
           titleAndBody.includes(t) ||
           (t === 'frontend' && (titleAndBody.includes('css') || titleAndBody.includes('react') || titleAndBody.includes('html') || titleAndBody.includes('ui'))) ||
           (t === 'backend' && (titleAndBody.includes('api') || titleAndBody.includes('server') || titleAndBody.includes('database'))) ||
           (t === 'devops' && (titleAndBody.includes('docker') || titleAndBody.includes('deploy') || titleAndBody.includes('ci'))) ||
           (t === 'ai' && (titleAndBody.includes('ai') || titleAndBody.includes('ml') || titleAndBody.includes('model')));
  });
  if (interestMatch) score += 25;

  // 3. Experience level (max 20 pts)
  const exp = experienceLevel.toLowerCase();
  const hasGoodFirst = issueLabels.some(l => l.toLowerCase().includes('good first issue'));
  const hasHelpWanted = issueLabels.some(l => l.toLowerCase().includes('help wanted'));
  
  if (exp === 'beginner' && hasGoodFirst) score += 20;
  else if (exp === 'intermediate' && (hasGoodFirst || hasHelpWanted)) score += 15;
  else if (exp === 'advanced' && hasHelpWanted) score += 10;
  else if (hasGoodFirst) score += 10;

  // 4. Label bonus (max 15 pts)
  if (hasGoodFirst) score += 10;
  if (hasHelpWanted) score += 5;

  // 5. Recency bonus (max 15 pts)
  const daysSinceCreation = (Date.now() - new Date(issue.created_at).getTime()) / (1000 * 60 * 60 * 24);
  if (daysSinceCreation < 7) score += 15;
  else if (daysSinceCreation < 30) score += 10;
  else if (daysSinceCreation < 90) score += 5;

  return Math.min(100, Math.max(0, Math.round(score)));
}
```

const formatDate = (dateString: string) => {
  return new Intl.DateTimeFormat('en-US', {
    month: 'short', day: 'numeric', year: 'numeric',
  }).format(new Date(dateString));
};

And the `getTextColorForBg` helper:
```typescript
const getTextColorForBg = (hexColor: string) => {
  const r = parseInt(hexColor.substring(0, 2), 16) || 0;
  const g = parseInt(hexColor.substring(2, 4), 16) || 0;
  const b = parseInt(hexColor.substring(4, 6), 16) || 0;
  const yiq = ((r * 299) + (g * 587) + (b * 114)) / 1000;
  return yiq >= 128 ? '#000000' : '#ffffff';
};
```
