Add the following handler functions to the component body (after existing hooks, before the return statement). These are the same logic as Git-Issue page.tsx lines 228-285, adapted to IssuesSwipe:

```typescript
const toggleLabel = (labelName: string) => {
  setExcludedLabels(prev => prev.filter(l => l !== labelName));
  setActiveLabels(prev =>
    prev.includes(labelName)
      ? prev.filter(l => l !== labelName)
      : [...prev, labelName]
  );
};

const toggleExcludeLabel = (labelName: string) => {
  setActiveLabels(prev => prev.filter(l => l !== labelName));
  setExcludedLabels(prev =>
    prev.includes(labelName)
      ? prev.filter(l => l !== labelName)
      : [...prev, labelName]
  );
};

const toggleLanguage = (langName: string) => {
  setActiveLanguages(prev =>
    prev.includes(langName)
      ? prev.filter(l => l !== langName)
      : [...prev, langName]
  );
};

const handleCustomLabelAdd = (e: React.KeyboardEvent<HTMLInputElement>) => {
  if (e.key === 'Enter' && customLabelInput.trim()) {
    const tags = customLabelInput.split(',').map(s => s.trim()).filter(Boolean);
    if (customLabelExcludeMode) {
      setExcludedLabels(prev => {
        const next = [...prev];
        tags.forEach(t => { if (!next.includes(t)) next.push(t); });
        return next;
      });
    } else {
      setActiveLabels(prev => {
        const next = [...prev];
        tags.forEach(t => { if (!next.includes(t)) next.push(t); });
        return next;
      });
    }
    setCustomLabelInput('');
  }
};

const clearAllFilters = () => {
  setActiveLabels([]);
  setExcludedLabels([]);
  setActiveLanguages([]);
  setUnassignedFilter(false);
  setNoCommentsFilter(false);
};

const hasActiveFilters =
  activeLabels.length > 0 || excludedLabels.length > 0 ||
  activeLanguages.length > 0 || unassignedFilter || noCommentsFilter;
```
